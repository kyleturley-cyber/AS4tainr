# Golden: what verify2 left open. The SQL message IDs of the statements
# that failed there only as SQL9010 (RUNSQLSTM ERRLVL(30) lists them); a
# RUNSQLSTM without errors (its listing and completion message); COMMIT
# and ROLLBACK under COMMIT(*CHG); the temporal table journaled, so that
# UPDATE and DELETE work and the history rows can be checked; DSPFD
# without OUTFILE, DSPFD TYPE(*ATR) FILEATR(*PF) to an outfile (QAFDPHY);
# a member's last-used date after it is read.
# Capture with: golden.ps1 -List tools\golden\verify3.txt -Files tools\golden\verify3
x_cleanup|!system -q "RUNSQL SQL('DROP TABLE AS4TRAINR1/GLDTMP') COMMIT(*NONE)"; for f in GLDTMP_HST GLDTO GLDTOV GLDATR GLDMBR GLDWSRC; do system -q "DLTF FILE(AS4TRAINR1/$f)"; done; system -q "DLTJRN JRN(AS4TRAINR1/GLDJRN)"; system -q "DLTJRNRCV JRNRCV(AS4TRAINR1/GLDRCV) DLTOPT(*IGNINQMSG)"; echo cleaned
x_tag|!for f in /home/AS4TRAINR/gld/gldtmp3.sql /home/AS4TRAINR/gld/glderr.sql /home/AS4TRAINR/gld/gldok.sql /home/AS4TRAINR/gld/gldcmt.sql; do setccsid 819 $f; done; ls /home/AS4TRAINR/gld | wc -l
x_out_table|RUNSQL SQL('CREATE TABLE AS4TRAINR1/GLDTO (SEQ INTEGER GENERATED ALWAYS AS IDENTITY, ID CHAR(10), VAL VARCHAR(1000))') COMMIT(*NONE)
x_script|RUNSQLSTM SRCSTMF('/home/AS4TRAINR/gld/gldtmp3.sql') COMMIT(*NONE)
x_errors|RUNSQLSTM SRCSTMF('/home/AS4TRAINR/gld/glderr.sql') COMMIT(*NONE) ERRLVL(30)
x_ok|RUNSQLSTM SRCSTMF('/home/AS4TRAINR/gld/gldok.sql') COMMIT(*NONE)
# Journaled: UPDATE and DELETE of the versioned table, then its history.
x_jrnrcv|CRTJRNRCV JRNRCV(AS4TRAINR1/GLDRCV)
x_jrn|CRTJRN JRN(AS4TRAINR1/GLDJRN) JRNRCV(AS4TRAINR1/GLDRCV)
x_strjrn|STRJRNPF FILE(AS4TRAINR1/GLDTMP AS4TRAINR1/GLDTMP_HST AS4TRAINR1/GLDTO) JRN(AS4TRAINR1/GLDJRN) IMAGES(*BOTH)
x_upd|!sleep 2; system "RUNSQL SQL('UPDATE AS4TRAINR1/GLDTMP SET CMD = ''WRKACTJOB'' WHERE GRP = ''G1''') COMMIT(*NONE)"
x_del|!sleep 2; system "RUNSQL SQL('DELETE FROM AS4TRAINR1/GLDTMP WHERE GRP = ''G1''') COMMIT(*NONE)"
x_hist|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLDTO (ID, VAL) SELECT ''HIST'', RTRIM(CMD) CONCAT '' '' CONCAT AUDIT_OPER CONCAT '' '' CONCAT CASE WHEN SYS_START = SYS_END THEN ''EQ'' WHEN SYS_START < SYS_END THEN ''LT'' ELSE ''GT'' END CONCAT '' '' CONCAT CASE WHEN TS_ID = SYS_START THEN ''TS=START'' ELSE ''TS<>START'' END FROM AS4TRAINR1/GLDTMP_HST ORDER BY SYS_START, SYS_END') COMMIT(*NONE)
x_chain|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLDTO (ID, VAL) SELECT ''CHAIN'', CASE WHEN EXISTS (SELECT 1 FROM AS4TRAINR1/GLDTMP_HST A, AS4TRAINR1/GLDTMP_HST B WHERE A.AUDIT_OPER = ''I'' AND B.AUDIT_OPER = ''U'' AND A.SYS_END = B.SYS_START) THEN ''END=NEXT START'' ELSE ''NO MATCH'' END FROM SYSIBM/SYSDUMMY1') COMMIT(*NONE)
x_asof|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLDTO (ID, VAL) SELECT ''ASOF'', RTRIM(GRP) CONCAT '' '' CONCAT CMD FROM AS4TRAINR1/GLDTMP FOR SYSTEM_TIME AS OF ''2000-01-01-00.00.00'' UNION ALL SELECT ''ASOFALL'', CHAR(COUNT(*)) FROM AS4TRAINR1/GLDTMP FOR SYSTEM_TIME FROM ''2000-01-01-00.00.00'' TO ''9999-12-30-00.00.00''') COMMIT(*NONE)
# Under commitment control: which of CMT1, CMT2, CMT3 stay.
x_cmt|RUNSQLSTM SRCSTMF('/home/AS4TRAINR/gld/gldcmt.sql') COMMIT(*CHG)
# DSPFD outfiles.
x_fd_nofile|DSPFD FILE(AS4TRAINR1/GLDTO) TYPE(*MBRLIST) OUTPUT(*OUTFILE)
x_fd_phy|DSPFD FILE(AS4TRAINR1/GLDTO) TYPE(*ATR) FILEATR(*PF) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLDATR)
x_fd_phyvals|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLDTO (ID, VAL) SELECT ''PHY'', PHFILE CONCAT ''|'' CONCAT PHLIB CONCAT ''|'' CONCAT PHFILA CONCAT ''|'' CONCAT PHDTAT CONCAT ''|'' CONCAT CHAR(PHNOFM) CONCAT ''|'' CONCAT CHAR(PHNOMB) CONCAT ''|'' CONCAT CHAR(PHMAXM) CONCAT ''|'' CONCAT CHAR(PHSIZ) CONCAT ''|'' CONCAT CHAR(PHMXRL) CONCAT ''|'' CONCAT PHJRNL CONCAT ''|'' CONCAT PHJRNM CONCAT ''|'' CONCAT PHJRLB CONCAT ''|'' CONCAT PHJRIM CONCAT ''|'' CONCAT PHSQLT FROM AS4TRAINR1/GLDATR') COMMIT(*NONE)
# A member's last-used date and count after it is read.
x_src|!system -q "CRTSRCPF FILE(AS4TRAINR1/GLDWSRC) RCDLEN(92)"; system -q "ADDPFM FILE(AS4TRAINR1/GLDWSRC) MBR(A1) SRCTYPE(SQL)"; system -q "ADDPFM FILE(AS4TRAINR1/GLDWSRC) MBR(B2) SRCTYPE(SQL)"; system "CPYFRMSTMF FROMSTMF('/home/AS4TRAINR/gld/gldok.sql') TOMBR('/QSYS.LIB/AS4TRAINR1.LIB/GLDWSRC.FILE/A1.MBR') MBROPT(*REPLACE)"; system "CPYF FROMFILE(AS4TRAINR1/GLDWSRC) TOFILE(*PRINT) FROMMBR(A1)" | wc -l
x_mbrl|DSPFD FILE(AS4TRAINR1/GLDWSRC) TYPE(*MBRLIST) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLDMBR)
x_mbrl_vals|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLDTO (ID, VAL) SELECT ''MBRL'', MLNAME CONCAT ''|'' CONCAT MLUCEN CONCAT ''|'' CONCAT MLUDAT CONCAT ''|'' CONCAT CHAR(MLUCNT) CONCAT ''|'' CONCAT CHAR(MLSIZE) FROM AS4TRAINR1/GLDMBR ORDER BY MLNAME') COMMIT(*NONE)
x_out|CPYF FROMFILE(AS4TRAINR1/GLDTO) TOFILE(*PRINT)
