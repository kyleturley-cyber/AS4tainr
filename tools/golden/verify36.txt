# values by default, with RMVBLANK(*TRAILING), ADDCOLNAM(*SQL), FLDDLM(';')
# and DECPNT(*COMMA), read back through CPYFRMSTMF into a one-field file.
# Capture with: .\tools\golden\golden.ps1 -List tools\golden\verify36.txt
b36_cleanup|!for c in "DLTF FILE(AS4TRAINR1/GLD36T)" "DLTF FILE(AS4TRAINR1/GLD36L)"; do system -q "$c"; done; echo cleaned
