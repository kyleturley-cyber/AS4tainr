# Golden: DSPPGMREF of an RPG program (a file and EXTPGM calls) printed
# and OUTPUT(*OUTFILE), and QADSPPGM's values for a CL program; DSPFD
# TYPE(*MBR) of an SQL index. (verify45's RPG source had no CCSID tag:
# it could not be opened.)
# Capture with: .\tools\golden\golden.ps1 -List tools\golden\verify46.txt -Files tools\golden\verify46
o46_cleanup|!for c in "DLTPGM PGM(AS4TRAINR1/GLD46G)" "DLTPGM PGM(AS4TRAINR1/GLD46C)" "DLTF FILE(AS4TRAINR1/GLD46I)" "DLTF FILE(AS4TRAINR1/GLD46P)" "DLTF FILE(AS4TRAINR1/GLD46S)" "DLTF FILE(AS4TRAINR1/GLD46R)" "DLTF FILE(AS4TRAINR1/GLD46V)"; do system -q "$c"; done; echo cleaned
o46_tag|!for f in gld46g.rpgle gld46p.pf gld46c.clle; do setccsid 819 /home/AS4TRAINR/gld/$f; done; ls /home/AS4TRAINR/gld | grep gld46
o46_src|!system -q "CRTSRCPF FILE(AS4TRAINR1/GLD46S) RCDLEN(92)"; system -q "ADDPFM FILE(AS4TRAINR1/GLD46S) MBR(GLD46P) SRCTYPE(PF)"; system "CPYFRMSTMF FROMSTMF('/home/AS4TRAINR/gld/gld46p.pf') TOMBR('/QSYS.LIB/AS4TRAINR1.LIB/GLD46S.FILE/GLD46P.MBR') MBROPT(*REPLACE)"; system "CPYFRMSTMF FROMSTMF('/home/AS4TRAINR/gld/gld46c.clle') TOMBR('/QSYS.LIB/AS4TRAINR1.LIB/GLD46S.FILE/GLD46C.MBR')"
o46_crtpf|!system -s "CRTPF FILE(AS4TRAINR1/GLD46P) SRCFILE(AS4TRAINR1/GLD46S)"
o46_crtc|CRTBNDCL PGM(AS4TRAINR1/GLD46C) SRCFILE(AS4TRAINR1/GLD46S) SRCMBR(GLD46C)
o46_crtg|CRTBNDRPG PGM(AS4TRAINR1/GLD46G) SRCSTMF('/home/AS4TRAINR/gld/gld46g.rpgle')
o46_pgmref|DSPPGMREF PGM(AS4TRAINR1/GLD46G) OUTPUT(*PRINT)
o46_refg|!system "DSPPGMREF PGM(AS4TRAINR1/GLD46G) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLD46R)" && system "CPYF FROMFILE(AS4TRAINR1/GLD46R) TOFILE(*PRINT) OUTFMT(*CHAR)"
o46_refc|!system "DSPPGMREF PGM(AS4TRAINR1/GLD46C) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLD46R)" && system "CPYF FROMFILE(AS4TRAINR1/GLD46R) TOFILE(*PRINT) OUTFMT(*CHAR)"
o46_refcv|!system -q "DLTF FILE(AS4TRAINR1/GLD46V)"; system "RUNSQL SQL('CREATE TABLE AS4TRAINR1/GLD46V AS (SELECT WHFNAM, WHLNAM, WHOTYP, WHOBJT, WHFUSG, WHRFNO, WHSPKG, WHRFNB, WHFNUM, WHSNAM FROM AS4TRAINR1/GLD46R) WITH DATA') COMMIT(*NONE)" && system "CPYF FROMFILE(AS4TRAINR1/GLD46V) TOFILE(*PRINT) OUTFMT(*CHAR)"
o46_index|RUNSQL SQL('CREATE INDEX AS4TRAINR1/GLD46I ON AS4TRAINR1/GLD46P (NAME)') COMMIT(*NONE)
o46_mbri|DSPFD FILE(AS4TRAINR1/GLD46I) TYPE(*MBR)
o46_end|!for c in "DLTPGM PGM(AS4TRAINR1/GLD46G)" "DLTPGM PGM(AS4TRAINR1/GLD46C)" "DLTF FILE(AS4TRAINR1/GLD46I)" "DLTF FILE(AS4TRAINR1/GLD46P)" "DLTF FILE(AS4TRAINR1/GLD46S)" "DLTF FILE(AS4TRAINR1/GLD46R)" "DLTF FILE(AS4TRAINR1/GLD46V)"; do system -q "$c"; done; echo cleaned
