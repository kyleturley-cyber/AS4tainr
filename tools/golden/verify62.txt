# Golden (batch 1 and 2): CPYF record selection (FROMKEY/TOKEY, NBRRCDS,
# INCCHAR, INCREL) on a keyed DDS file: each copy's completion message and
# the records copied. Command DEP statements without MSGID: what a command
# whose parameters break a dependency gives. XML-INTO %HANDLER: the
# handler's totals.
# Capture with: .\tools\golden\golden.ps1 -List tools\golden\verify62.txt -Files tools\golden\verify62
e62_cleanup|!for c in "DLTF FILE(AS4TRAINR1/GLD62C)" "DLTF FILE(AS4TRAINR1/GLD62S)" "DLTF FILE(AS4TRAINR1/GLD62O)" "DLTCMD CMD(AS4TRAINR1/GLD62D)" "DLTPGM PGM(AS4TRAINR1/GLD62P)" "DLTPGM PGM(AS4TRAINR1/GLD62X)" "DLTF FILE(AS4TRAINR1/GLD62R)"; do system -q "$c"; done; echo cleaned
e62_tag|!for f in gld62c.pf gld62d.cmd gld62p.clle gld62x.rpgle; do setccsid 819 /home/AS4TRAINR/gld/$f; done; ls /home/AS4TRAINR/gld | grep gld62
e62_make|!system -q "CRTSRCPF FILE(AS4TRAINR1/GLD62S) RCDLEN(92)"; system "CPYFRMSTMF FROMSTMF('/home/AS4TRAINR/gld/gld62c.pf') TOMBR('/QSYS.LIB/AS4TRAINR1.LIB/GLD62S.FILE/GLD62C.MBR')"; system "CRTPF FILE(AS4TRAINR1/GLD62C) SRCFILE(AS4TRAINR1/GLD62S)" | grep -E "^CP[CF]"; system "RUNSQL SQL('INSERT INTO AS4TRAINR1/GLD62C VALUES(3, ''Bluebird Supply'', ''Omaha'', 0), (1, ''Acme Tooling'', ''Des Moines'', 1250.50), (5, ''Lake Pine'', ''Duluth'', -75.25), (2, ''Midwest Fasteners'', ''Chicago'', 310), (4, ''Prairie Freight'', ''Lincoln'', 9999999.99)') COMMIT(*NONE)"; echo made
e62_key|!system "CPYF FROMFILE(AS4TRAINR1/GLD62C) TOFILE(AS4TRAINR1/GLD62O) CRTFILE(*YES) FROMKEY(1 2) TOKEY(1 4)"; system "CPYF FROMFILE(AS4TRAINR1/GLD62O) TOFILE(*PRINT)" | grep -E "^CP|Acme|Midwest|Bluebird|Prairie|Lake"
e62_bldkey|!system "CPYF FROMFILE(AS4TRAINR1/GLD62C) TOFILE(AS4TRAINR1/GLD62O) MBROPT(*REPLACE) FROMKEY(*BLDKEY (2)) NBRRCDS(2)"; system "CPYF FROMFILE(AS4TRAINR1/GLD62O) TOFILE(*PRINT)" | grep -E "^CP|Acme|Midwest|Bluebird|Prairie|Lake"
e62_increl|!system "CPYF FROMFILE(AS4TRAINR1/GLD62C) TOFILE(AS4TRAINR1/GLD62O) MBROPT(*REPLACE) INCREL((*IF CUSBAL *GT 0) (*AND CUSCTY *NE 'Chicago') (*OR CUSNO *EQ 5))"; system "CPYF FROMFILE(AS4TRAINR1/GLD62O) TOFILE(*PRINT)" | grep -E "^CP|Acme|Midwest|Bluebird|Prairie|Lake"
e62_incchar|!system "CPYF FROMFILE(AS4TRAINR1/GLD62C) TOFILE(AS4TRAINR1/GLD62O) MBROPT(*REPLACE) INCCHAR(CUSNAM 1 *EQ 'P')"; system "CPYF FROMFILE(AS4TRAINR1/GLD62O) TOFILE(*PRINT)" | grep -E "^CP|Acme|Midwest|Bluebird|Prairie|Lake"; system "CPYF FROMFILE(AS4TRAINR1/GLD62C) TOFILE(AS4TRAINR1/GLD62O) MBROPT(*REPLACE) INCCHAR(CUSNAM 1 *CT 'Supply')"; system "CPYF FROMFILE(AS4TRAINR1/GLD62O) TOFILE(*PRINT)" | grep -E "^CP|Acme|Midwest|Bluebird|Prairie|Lake"
e62_bad|!system "CPYF FROMFILE(AS4TRAINR1/GLD62C) TOFILE(AS4TRAINR1/GLD62O) MBROPT(*REPLACE) INCREL((*IF NOFIELD *EQ 1))"
e62_dep|!for m in GLD62D GLD62P; do f=$(echo $m | tr A-Z a-z); x=cmd; [ $m = GLD62P ] && x=clle; system "CPYFRMSTMF FROMSTMF('/home/AS4TRAINR/gld/$f.$x') TOMBR('/QSYS.LIB/AS4TRAINR1.LIB/GLD62S.FILE/$m.MBR')" > /dev/null; done; system "CRTBNDCL PGM(AS4TRAINR1/GLD62P) SRCFILE(AS4TRAINR1/GLD62S)" | grep -E "^CP[CF]"; system "CRTCMD CMD(AS4TRAINR1/GLD62D) PGM(AS4TRAINR1/GLD62P) SRCFILE(AS4TRAINR1/GLD62S)" | grep -E "^CP[CF]"; for c in "TYPE(*DTA)" "TYPE(*FILE)" "TYPE(*FILE) FILE(X)" "A(Y)" "A(Y) B(N)"; do echo "== $c"; system "AS4TRAINR1/GLD62D $c"; echo "rc=$?"; done
e62_handler|!system "RUNSQL SQL('CREATE TABLE AS4TRAINR1/GLD62R (RES CHAR(40) NOT NULL)') COMMIT(*NONE)"; system "CRTBNDRPG PGM(AS4TRAINR1/GLD62X) SRCSTMF('/home/AS4TRAINR/gld/gld62x.rpgle')" | grep -E "^RNS9304|^CPF"; system "CALL PGM(AS4TRAINR1/GLD62X)"; system "CPYF FROMFILE(AS4TRAINR1/GLD62R) TOFILE(*PRINT)" | grep -E "^ +[0-9]+ +[0-9]"
e62_end|!for c in "DLTF FILE(AS4TRAINR1/GLD62C)" "DLTF FILE(AS4TRAINR1/GLD62S)" "DLTF FILE(AS4TRAINR1/GLD62O)" "DLTCMD CMD(AS4TRAINR1/GLD62D)" "DLTPGM PGM(AS4TRAINR1/GLD62P)" "DLTPGM PGM(AS4TRAINR1/GLD62X)" "DLTF FILE(AS4TRAINR1/GLD62R)"; do system -q "$c"; done; echo cleaned
