# Golden: verify6 found nothing in GLD6LOG after 20 seconds: run the
# batch job again and wait longer, then show its status, its job log
# and its spooled files (which printer file each display command uses).
# CPYF OUTFMT(*HEX) of a character and a packed field.
# Capture with: golden.ps1 -List tools\golden\verify7.txt -Files tools\golden\verify7
c7_cleanup|!system -q "DLTPGM PGM(AS4TRAINR1/GLD6SPL)"; for f in GLD6TB GLD6SP GLD6LOG GLD7HX; do system -q "DLTF FILE(AS4TRAINR1/$f)"; done; echo cleaned
c7_tag|!setccsid 819 /home/AS4TRAINR/gld/gld6spl.clle; ls /home/AS4TRAINR/gld/gld6spl.clle
c7_table|RUNSQL SQL('CREATE TABLE AS4TRAINR1/GLD6TB (ID CHAR(10) NOT NULL, VAL CHAR(60) NOT NULL)') COMMIT(*NONE)
c7_log|RUNSQL SQL('CREATE TABLE AS4TRAINR1/GLD6LOG (L VARCHAR(300) NOT NULL)') COMMIT(*NONE)
c7_sp|CRTPF FILE(AS4TRAINR1/GLD6SP) RCDLEN(133)
c7_crt|!system -s "CRTBNDCL PGM(AS4TRAINR1/GLD6SPL) SRCSTMF('/home/AS4TRAINR/gld/gld6spl.clle')"
c7_sbm|SBMJOB CMD(CALL PGM(AS4TRAINR1/GLD6SPL)) JOB(GLD7SPL)
c7_wait|!sleep 90; echo waited
c7_jobs|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLD6LOG SELECT ''JOB '' CONCAT JOB_NAME CONCAT '' '' CONCAT COALESCE(JOB_STATUS, ''-'') FROM TABLE(QSYS2.JOB_INFO(JOB_USER_FILTER => ''AS4TRAINR'')) X WHERE JOB_NAME LIKE ''%/GLD7SPL''') COMMIT(*NONE)
c7_spl|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLD6LOG SELECT ''SPL '' CONCAT SPOOLED_FILE_NAME CONCAT '' '' CONCAT COALESCE(USER_DATA, ''-'') FROM QSYS2.OUTPUT_QUEUE_ENTRIES_BASIC WHERE JOB_NAME LIKE ''%/GLD7SPL'' ORDER BY CREATE_TIMESTAMP') COMMIT(*NONE)
c7_joblog|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLD6LOG SELECT ''LOG '' CONCAT COALESCE(M.MESSAGE_ID, ''-'') CONCAT '' '' CONCAT COALESCE(VARCHAR(M.MESSAGE_TEXT, 200), '''') FROM TABLE(QSYS2.JOB_INFO(JOB_USER_FILTER => ''AS4TRAINR'')) J, TABLE(QSYS2.JOBLOG_INFO(J.JOB_NAME)) M WHERE J.JOB_NAME LIKE ''%/GLD7SPL''') COMMIT(*NONE)
c7_msgs|CPYF FROMFILE(AS4TRAINR1/GLD6LOG) TOFILE(*PRINT)
c7_lines|CPYF FROMFILE(AS4TRAINR1/GLD6SP) TOFILE(*PRINT)
c7_two|CPYF FROMFILE(AS4TRAINR1/GLD6TB) TOFILE(*PRINT)
c7_hx_table|RUNSQL SQL('CREATE TABLE AS4TRAINR1/GLD7HX (ID CHAR(4) NOT NULL, N DECIMAL(5, 0) NOT NULL)') COMMIT(*NONE)
c7_hx_row|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLD7HX VALUES(''AB'', 123), (''Z9'', -45)') COMMIT(*NONE)
c7_hex|CPYF FROMFILE(AS4TRAINR1/GLD7HX) TOFILE(*PRINT) OUTFMT(*HEX)
c7_char|CPYF FROMFILE(AS4TRAINR1/GLD7HX) TOFILE(*PRINT)
c7_end|!system -q "DLTPGM PGM(AS4TRAINR1/GLD6SPL)"; for f in GLD6TB GLD6SP GLD6LOG GLD7HX; do system -q "DLTF FILE(AS4TRAINR1/$f)"; done; echo cleaned
