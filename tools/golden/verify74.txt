# Golden (batch 17): DSPMOD's output files (Code for IBM i's getModuleExports:
# DSPMOD DETAIL(*EXPORT) OUTPUT(*OUTFILE), fields EXLBNM, EXMONM, EXMOAT,
# EXSYNM, EXSYTY, EXOPPP), their layouts (DSPFFD) and rows for a CL module,
# DETAIL(*BASIC) and *IMPORT; DSPMOD's printed report.
u74_cleanup|!system -q "DLTMOD MODULE(AS4TRAINR1/GLD74M)"; for f in GLD74X GLD74B GLD74I; do system -q "DLTF FILE(AS4TRAINR1/$f)"; done; rm -f /tmp/gld74.clle; echo cleaned
u74_make|!printf 'PGM\r\nDCL VAR(&X) TYPE(*CHAR) LEN(5)\r\nCALL PGM(QCMDEXC) PARM(X 1)\r\nENDPGM\r\n' > /tmp/gld74.clle; setccsid 819 /tmp/gld74.clle; system "CRTCLMOD MODULE(AS4TRAINR1/GLD74M) SRCSTMF('/tmp/gld74.clle')"
u74_exp|DSPMOD MODULE(AS4TRAINR1/GLD74M) DETAIL(*EXPORT) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLD74X)
u74_expffd|DSPFFD FILE(AS4TRAINR1/GLD74X)
u74_exprows|!/QOpenSys/usr/bin/qsh -c "db2 \"SELECT * FROM AS4TRAINR1.GLD74X\""
u74_bas|DSPMOD MODULE(AS4TRAINR1/GLD74M) DETAIL(*BASIC) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLD74B)
u74_basffd|DSPFFD FILE(AS4TRAINR1/GLD74B)
u74_basrows|!/QOpenSys/usr/bin/qsh -c "db2 \"SELECT * FROM AS4TRAINR1.GLD74B\""
u74_imp|DSPMOD MODULE(AS4TRAINR1/GLD74M) DETAIL(*IMPORT) OUTPUT(*OUTFILE) OUTFILE(AS4TRAINR1/GLD74I)
u74_impffd|DSPFFD FILE(AS4TRAINR1/GLD74I)
u74_improws|!/QOpenSys/usr/bin/qsh -c "db2 \"SELECT * FROM AS4TRAINR1.GLD74I\""
u74_print|DSPMOD MODULE(AS4TRAINR1/GLD74M) DETAIL(*BASIC *EXPORT *IMPORT) OUTPUT(*PRINT)
u74_end|!system -q "DLTMOD MODULE(AS4TRAINR1/GLD74M)"; for f in GLD74X GLD74B GLD74I; do system -q "DLTF FILE(AS4TRAINR1/$f)"; done; rm -f /tmp/gld74.clle; echo cleaned
