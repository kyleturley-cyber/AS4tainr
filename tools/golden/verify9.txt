d9_prt|!for c in "DSPOBJD OBJ(AS4TRAINR1/GLD9PF) OBJTYPE(*FILE) OUTPUT(*PRINT)" "DSPFD FILE(AS4TRAINR1/GLD9PF) OUTPUT(*PRINT)" "DSPFFD FILE(AS4TRAINR1/GLD9PF) OUTPUT(*PRINT)" "DSPDBR FILE(AS4TRAINR1/GLD9PF) OUTPUT(*PRINT)" "DSPLIBL OUTPUT(*PRINT)" "DSPSYSVAL SYSVAL(QDATE) OUTPUT(*PRINT)" "DSPMSGD RANGE(CPF9898) MSGF(QCPFMSG) OUTPUT(*PRINT)" "DSPJOB OUTPUT(*PRINT) OPTION(*DFNA)" "DSPLIB LIB(AS4TRAINR1) OUTPUT(*PRINT)" "CPYF FROMFILE(AS4TRAINR1/GLD9PF) TOFILE(*PRINT)" "DSPUSRPRF USRPRF(AS4TRAINR) OUTPUT(*PRINT)" "DSPPGMREF PGM(AS4TRAINR1/*ALL) OUTPUT(*PRINT)"; do system -k -q "$c" >/dev/null 2>&1; echo "$c" | cut -d' ' -f1; done
d9_prt_list|RUNSQL SQL('INSERT INTO AS4TRAINR1/GLD9LOG SELECT CHAR(SPOOLED_FILE_NAME CONCAT '' '' CONCAT COALESCE(USER_DATA, ''-'') CONCAT '' '' CONCAT JOB_NAME, 60) FROM QSYS2.OUTPUT_QUEUE_ENTRIES_BASIC WHERE USER_NAME = ''AS4TRAINR'' ORDER BY CREATE_TIMESTAMP DESC FETCH FIRST 15 ROWS ONLY') COMMIT(*NONE)
d9_prt_rows|CPYF FROMFILE(AS4TRAINR1/GLD9LOG) TOFILE(*PRINT)
d9_end|!for f in GLD9LF GLD9PF GLD9C GLD9P GLD9SRC GLD9SP GLD9V GLD9DBR GLD9BAS GLD9MBR GLD9FMT GLD9ACP GLD9BL GLD9AA GLD9TW GLD9J GLD9LOG; do system -q "DLTF FILE(AS4TRAINR1/$f)"; done; echo cleaned
